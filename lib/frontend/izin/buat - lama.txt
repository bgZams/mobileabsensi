Future<void> kirimizin() async {
    String tanggalTerpilih = tanggal.text;
    String keteranganValue = keterangan.text;
    String durasiValue = durasi.text;
    String imagePath = image!.path;
    String fileName = imagePath.split('/').last;

    Map<String, dynamic> addFirebaseIzin = {
      'id_user': iduser,
      'id_atasan': iduserpimpinan,
      'id_status': 0,
      'nama_lengkap': namalengkap,
      'jenis_izin': _valJenisIzin,
      'key_notif': 'izin'
    }; 
    print(iduserpimpinan);
    try {
      // Define a function to send the request
      Future<void> sendRequest(String endpoint) async {
        final response = await http.get(
          Uri.parse(endpoint),
          headers: {
            'Content-type': 'application/json',
            'Accept': 'application/json'
          },
        );
        final data = jsonDecode(response.body);
          String message = json.encode(data["message"]).replaceAll('"', '');

        if (response.statusCode == 200) {
          
          if (data['code'] == 2 || data['code'] == 3 || data['code'] == 4) {
              // ignore: use_build_context_synchronously
                                        QuickAlert.show(
                                                context: context,
                                                type: QuickAlertType.warning,
                                                text: message,
                                                );  
          } else {
            // await FTPServices().uploadFileToServer(imagePath, fileName);
            if (data['code'] == 1) {
              final DatabaseReference databaseReference =
                  FirebaseDatabase.instance.reference();
              databaseReference.child("izin").push().set(addFirebaseIzin);
               // ignore: use_build_context_synchronously
                                        QuickAlert.show(
                                                context: context,
                                                type: QuickAlertType.success,
                                                text: message,
                                                );  
              resetState();
            }
          }
        }
      }
        await sendRequest(
            '$url/api_android/api_flutter/kirim_izin.php?val_izin=$_valJenisIzin&id_user=$iduser&id_atasan=$iduserpimpinan&username=$username&nama_lengkap=$namalengkap&id_instansi=$idinstansi&tglawal=$tanggalTerpilih&id_keterangan=0&keterangan=$keteranganValue&durasi=$durasiValue&id_admin_instansi=$idadmininstansi&namafile=$fileName');
    } catch (e) {
      showServerErrorDialog();
    }
  }
